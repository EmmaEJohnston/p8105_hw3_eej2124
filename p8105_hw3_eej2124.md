demographic_df =
  read_csv("data/nhanes_demographics.csv",
           skip = 4,

  mutate(
    sex = case_match(

     2 ~ c("female")), 
    education = case_match(

      1 ~ c("less than high school"), 
      2 ~ c("high school equivalent"),
